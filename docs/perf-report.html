Decision rule

1. 전송 바이트를 먼저 줄입니다.

2. 직렬 네트워크 대기를 병렬 풀로 바꿉니다.

3. SQLite는 속도보다 내구성 문제로 다룹니다.

4. 측정상 미미한 최적화는 뒤로 보냅니다.

import zlib from "node:zlib";

/** 1KB 이상 텍스트 응답만 gzip. 로컬 단일 사용자라 br(느림) 대신 gzip 고정. */
function gzipMiddleware(): express.RequestHandler {
  const MIN = 1024;
  return (req, res, next) => {
    if (!/\bgzip\b/.test(req.headers["accept-encoding"] ?? "")) return next();
    const json = res.json.bind(res);
    res.json = (body: unknown) => {
      const buf = Buffer.from(JSON.stringify(body));
      if (buf.length < MIN) return json(body);
      const gz = zlib.gzipSync(buf, { level: 6 });
      res.setHeader("Content-Type", "application/json; charset=utf-8");
      res.setHeader("Content-Encoding", "gzip");
      res.setHeader("Vary", "Accept-Encoding");
      res.removeHeader("Content-Length");
      return res.end(gz);
    };
    next();
  };
}

export function createApp() {
  const app = express();
  app.use(gzipMiddleware());
  app.use(express.json());
  app.use("/api", api);
}

정적 자산까지 함께 처리할 경우, tech-lead 승인 후 compression 패키지를 쓰는 방식이 가장 단순합니다.

src/stock/holdings.ts
const QUOTE_TTL_MS = 5 * 60_000;
const quoteMemo = new Map<string, { at: number; points: StockPoint[] }>();

async function pointsFor(h: StockHolding): Promise<StockPoint[]> {
  const key = `${h.market}|${h.quoteCode}`;
  const hit = quoteMemo.get(key);
  if (hit && Date.now() - hit.at < QUOTE_TTL_MS) return hit.points;
  try {
    const fresh = await fetchHistory(h.quoteCode, h.market, HISTORY_DAYS);
    if (fresh.length) {
      quoteMemo.set(key, { at: Date.now(), points: fresh });
      return fresh;
    }
  } catch (e) {
    log.warn(`보유 종목 시세 조회 실패 ${h.name}(${h.symbol}): ${(e as Error).message}`);
  }
  const ticker = getStockBySymbol(h.market, h.symbol);
  return ticker ? getStockPoints(ticker.id, HISTORY_DAYS) : [];
}

export async function summarizeHoldings(market?: StockMarket) {
  const holdings = listHoldings(market);
  return mapWithLimit(holdings, 3, async (h) => {
    const points = await pointsFor(h);
    const last = points.length ? points[points.length - 1] : null;
    return summarizeHolding(h, {
      close: last?.close ?? null,
      changePct: last?.changePct ?? null,
      currency: last?.currency ?? (h.market === "kr" ? "KRW" : "USD"),
    }, points);
  });
}

let memo: ChannelCache | null = null;
let dirty = false;
let flushTimer: NodeJS.Timeout | null = null;

function cache(): ChannelCache {
  if (!memo) memo = readCache();
  return memo;
}

function markDirty(): void {
  dirty = true;
  if (flushTimer) return;
  flushTimer = setTimeout(() => {
    flushTimer = null;
    if (!dirty || !memo) return;
    dirty = false;
    writeCache(memo);
  }, 1000);
  flushTimer.unref?.();
}

export async function resolveChannelId(query: string): Promise<string | null> {
  const key = cacheKey(query);
  if (!key) return null;
  const c = cache();
  const hit = c[key];
  if (hit) {
    if (hit.channelId) return hit.channelId;
    const age = Date.now() - new Date(hit.resolvedAt).getTime();
    if (Number.isFinite(age) && age < NEG_TTL_MS) return null;
  }
  const id = await resolveUncached(query);
  c[key] = { channelId: id, resolvedAt: new Date().toISOString() };
  markDirty();
  return id;
}

src/util/pool.ts
export async function mapWithLimit<T, R>(
  items: readonly T[], limit: number, fn: (item: T, i: number) => Promise<R>
): Promise<R[]> {
  const out = new Array<R>(items.length);
  let next = 0;
  const workers = Array.from({ length: Math.min(limit, items.length) }, async () => {
    for (;;) {
      const i = next++;
      if (i >= items.length) return;
      out[i] = await fn(items[i], i);
    }
  });
  await Promise.all(workers);
  return out;
}

카테고리별 query를 합쳐 전역 중복을 제거한 뒤 동시성 6의 공용 풀에서 harvestOneChannel을 실행하고, 결과를 카테고리에 다시 배분합니다.

기대 효과 · 84 KB 파싱 100회 → 1회, 84 KB 쓰기 N회 → 디바운스 1회. 카테고리 수 5만큼 누적되던 벽시계 시간과 lost update 결함을 함께 줄입니다.

메모리 캐시는 파일 수동 수정이 재시작 전 반영되지 않습니다. 프로세스 종료 시 디바운스 플러시가 유실되지 않도록 process.on("beforeExit") 훅이 필요합니다.

const stmts = new Map<string, StatementSync>();

export function stmt(sql: string): StatementSync {
  let s = stmts.get(sql);
  if (!s) {
    s = db().prepare(sql);
    stmts.set(sql, s);
  }
  return s;
}

export function closeDb(): void {
  if (_db) {
    stmts.clear();
    _db.close();
    _db = null;
  }
}
